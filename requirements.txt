# Production dependencies only; every line is == pinned so Docker/Render rebuilds are reproducible.
# Dev/test tools live in requirements-dev.txt so the runtime image stays small.

fastapi==0.162.0
# ASGI server; [standard] adds uvloop/httptools for faster request handling on Linux

# Feature arrays for model input; numpy<2 cannot unpickle the mexico/qatar models

# CPU-only wheel on Linux skips the bundled CUDA libs, shrinking the container image

# Windows/macOS dev boxes have no xgboost-cpu need, so they take the regular wheel
xgboost==3.2.0; sys_platform != "linux"

joblib==1.6.0
